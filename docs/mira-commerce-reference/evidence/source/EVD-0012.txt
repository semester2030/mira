14|@Injectable()
15|export class PartnerTokenGuard implements CanActivate {
16|  constructor(private readonly prisma: PrismaService) {}
17|
18|  async canActivate(context: ExecutionContext): Promise<boolean> {
19|    const request = context.switchToHttp().getRequest<PartnerRequest>();
20|    const header = request.headers.authorization;
21|    if (!header?.startsWith('Bearer ')) {
22|      throw new UnauthorizedException('Missing partner token');
23|    }
24|    const token = header.slice(7).trim();
25|    const user = await this.prisma.partnerUser.findUnique({
26|      where: { accessToken: token },
27|      include: { partner: true },
28|    });
29|    if (!user || user.partner.status !== 'active') {
30|      throw new UnauthorizedException('Invalid or inactive partner token');
31|    }
32|    request.partnerUser = {
33|      id: user.id,
34|      partnerId: user.partnerId,
35|      email: user.email,
36|    };
37|    return true;
38|  }
